place(self, source: Path, output: Path, old: str, new: str) -> dict:
        if not old: raise ValueError("찾을 문자열이 비어 있습니다.")
        if source.resolve() == output.resolve(): raise ValueError("원본으로 덮어쓸 수 없습니다.")
        before_hash = sha256(source)
        before = self.parse(source)
        with ZipFile(source, "r") as zin:
            infos = zin.infolist(); patched = {}; occurrences = 0
            for info in infos:
                if not re.fullmatch(r"Contents/section\d+\.xml", info.filename, re.I): continue
                root = safe_xml(zin.read(info.filename)); count = 0
                for n in root.iter():
                    if lname(n) == "t" and n.text and old in n.text:
                        c = n.text.count(old); n.text = n.text.replace(old, new); count += c; occurrences += c
                if count: patched[info.filename] = etree.tostring(root, xml_declaration=True, encoding="UTF-8", standalone=True)
            if not occurrences: raise ValueError(f"'{old}' 문자열을 찾지 못했습니다.")
            output.parent.mkdir(parents=True, exist_ok=True)
            with ZipFile(output, "w") as zout:
                for info in infos: zout.writestr(info, patched.get(info.filename, zin.read(info.filename)))
        after = self.parse(output)
        return {
            "occurrences": occurrences, "sourceUnchanged": sha256(source) == before_hash,
            "structurePreserved": len(before["paragraphs"]) == len(after["paragraphs"]) and len(before["tables"]) == len(after["tables"]),
            "output": str(output),
        }
