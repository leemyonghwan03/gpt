            pass

    return None, "not-found"


def find_hwp_window() -> int | None:
    if sys.platform != "win32":
        return None
    import win32gui
    result = None

    def cb(hwnd, _):
        nonlocal result
        if result is not None or not win32gui.IsWindowVisible(hwnd):
            return
        title = (win32gui.GetWindowText(hwnd) or "").lower()
        if "한글" in title or ".hwp" in title or ".hwpx" in title:
            result = int(hwnd)

    win32gui.EnumWindows(cb, None)
    return result


def focus_hwp_window(hwnd: int) -> bool:
    try:
        import win32con
        import win32gui
        if win32gui.IsIconic(hwnd):
            win32gui.ShowWindow(hwnd, win32con.SW_RESTORE)
        win32gui.SetForegroundWindow(hwnd)
        return True
    except Exception:
        return False


def send_new_document(hwnd: int) -> bool:
    """Ctrl+N in Hangul: File > New / new document."""
    try:
        focus_hwp_window(hwnd)
        from pywinauto.keyboard import send_keys
        send_keys("^n", pause=0.05)
        return True
    except Exception:
        return False


def open_hancom(*, create_new_document: bool = False, force_new_process: bool = False) -> dict:
    """
    Open Hangul or create a blank document.
    - Existing Hangul + normal open: bring it forward.
    - Existing Hangul + new document: Ctrl+N.
    - No Hangul: start Hwp.exe, then Ctrl+N for explicit new document.
    """
    if sys.platform != "win32":
        return {"ok": False, "reason": "non-windows"}

    hwnd = find_hwp_window()
    if hwnd and not force_new_process:
        if create_new_document:
            ok = send_new_document(hwnd)
            return {"ok": ok, "reason": "ctrl-n-existing" if ok else "ctrl-n-failed", "hwnd": hwnd}
        focus_hwp_window(hwnd)
        return {"ok": True, "reason": "already-open", "hwnd": hwnd}

    exe, source = discover_hwp_exe()
    if not exe:
        return {"ok": False, "reason": "hwp-not-found", "discovery": source}

    try:
        proc = subprocess.Popen(
            [exe],
            cwd=str(Path(exe).parent),
            creationflags=getattr(subprocess, "CREATE_NEW_PROCESS_GROUP", 0),
        )
    except Exception as exc:
        return {"ok": False, "reason": "launch-failed", "error": f"{type(exc).__name__}: {exc}"}

    # Explicit new document: after Hwp has created a window, send Ctrl+N.
    if create_new_document:
        deadline = time.time() + 8.0
        target = None
        while time.time() < deadline:
            time.sleep(0.25)
            target = find_hwp_window()
            if target:
                break
        if target:
            send_new_document(target)

    return {"ok": True, "reason": "launched", "pid": proc.pid, "executable": exe, "discovery": source}
