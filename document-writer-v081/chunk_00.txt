def frozen_dir() -> Path:
    if getattr(sys, "frozen", False):
        return Path(sys.executable).resolve().parent
    return Path(__file__).resolve().parent


def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def safe_xml(data: bytes):
    return etree.fromstring(
        data,
        parser=etree.XMLParser(
            resolve_entities=False,
            no_network=True,
            load_dtd=False,
            huge_tree=False,
            recover=False,
        ),
    )


def lname(node) -> str:
    return etree.QName(node).localname


def validate_zip(path: Path) -> dict:
    with ZipFile(path) as zf:
        infos = zf.infolist()
        if len(infos) > 10000:
            raise ValueError("ZIP entry limit exceeded")
        total = 0
        seen = set()
        for info in infos:
            name = info.filename.replace("\\", "/")
            if name.startswith("/") or ".." in Path(name).parts:
                raise ValueError(f"unsafe ZIP path: {name}")
            if name in seen:
                raise ValueError(f"duplicate ZIP entry: {name}")
            seen.add(name)
            total += info.file_size
            if info.file_size > 128 * 1024 * 1024:
                raise ValueError(f"entry too large: {name}")
            if total > 512 * 1024 * 1024:
                raise ValueError("ZIP total size limit exceeded")
            if info.compress_size and info.file_size > 1024 * 1024:
                if info.file_size / info.compress_size > 200:
                    raise ValueError(f"suspicious compression ratio: {name}")
        bad = zf.testzip()
        if bad:
            raise ValueError(f"CRC failure: {bad}")
        return {"entries": len(infos), "names": [i.filename for i in infos]}

LABEL_RE = re.compile(r"^\s*([가-힣A-Za-z0-9_ /·().\-]{1,40}?)\s*[:：]\s*(.+?)\s*$")
DATE_RE = re.compile(r"^\d{4}[.\-/]\d{1,2}[.\-/]\d{1,2}\.?$")


def infer_type(label: str, value: str) -> str:
    compact = re.sub(r"\s+", "", label)
    if compact in KNOWN and KNOWN[compact] == "date":
        return "date"
    if DATE_RE.match(value.strip()):
        return "date"
    if re.match(r"^[+-]?\d[\d,]*(?:\.\d+)?(?:\s*(?:건|명|대|개|회|%|원))?$", value.strip()):
        return "number"
    if "\n" in value or len(value) > 120:
        return "multiline_text"
    return "text"
