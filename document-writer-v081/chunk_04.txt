    def open_hancom_ui(self):
        try:
            result = open_hancom(create_new_document=False)
            if result.get("ok"):
                self.status.set("한글을 열었습니다." if result.get("reason") == "launched" else "열려 있는 한글로 이동")
                return
            if result.get("reason") == "hwp-not-found":
                self.messagebox.showerror(APP_TITLE, "한글(Hwp.exe)을 찾지 못했습니다.\n한컴오피스 설치 상태를 확인하세요.")
                self.status.set("Hwp.exe를 찾지 못했습니다.")
                return
            raise RuntimeError(result.get("error") or str(result))
        except Exception as e:
            self.error(e)

    def apply_fields(self):
        try:
            p=self.require();out=self.filedialog.asksaveasfilename(initialdir=str(p.parent),initialfile=f"{datetime.now():%Y%m%d}_{p.stem}_작성본.hwpx",defaultextension=".hwpx",filetypes=[("HWPX","*.hwpx")]);
            if not out:return
            r=self.engine.write_fields(p,Path(out),{k:v.get() for k,v in self.field_vars.items()});
            if not all([r['sourceUnchanged'],r['valuesVerified'],r['structurePreserved']]):raise RuntimeError(str(r))
            self.show(f"새 문서 작성 완료\n{out}\n\n원본 보호 PASS · 값 검증 PASS · 구조 보존 PASS","작성 완료")
        except Exception as e:self.error(e)
