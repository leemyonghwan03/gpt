       raise ValueError(f"Text node 없음: {f.name}")
                        text_nodes[0].text = new_value
                        for n in text_nodes[1:]: n.text = ""
                    else:
                        replaced = False
                        for run in runs:
                            for n in run.iter():
                                if lname(n) == "t" and n.text and f.old_value in n.text:
                                    n.text = n.text.replace(f.old_value, new_value, 1)
                                    replaced = True
                                    break
                            if replaced: break
                        if not replaced:
                            raise ValueError(f"Field 치환 실패: {f.name}")
                patched[part] = etree.tostring(root, xml_declaration=True, encoding="UTF-8", standalone=True)
            output.parent.mkdir(parents=True, exist_ok=True)
            with ZipFile(output, "w") as zout:
                for info in infos:
                    zout.writestr(info, patched.get(info.filename, zin.read(info.filename)))

        validate_zip(output)
        after = self.parse(output)
        after_fields = {f.name: f.value for f in after["fields"]}
        return {
            "sourceUnchanged": sha256(source) == before_hash,
            "valuesVerified": all(after_fields.get(k) == str(v) for k, v in values.items() if k in field_map),
            "structurePreserved": (
                len(parsed["paragraphs"]) == len(after["paragraphs"])
                and len(parsed["tables"]) == len(after["tables"])
                and len(parsed["fonts"]) == len(after["fonts"])
                and len(parsed["charStyles"]) == len(after["charStyles"])
                and parsed["images"] == after["images"]
                and parsed["equations"] == after["equations"]
            ),
            "output": str(output), "changedParts": sorted(patched),
        }

    def batch_replace(self, source: Path, output: Path, old: str, new: str) -> dict:
        if not old: raise ValueError("찾을 문자열이 비어 있습니다.")
        if source.resolve() == output.resolve(): raise ValueError("원본으로 덮어쓸 수 없습니다.")
        before_hash = sha256(source)
        before = self.parse(source)
        with ZipFile(source, "r") as zin:
            infos = zin.infolist(); patched = {}; occurrences = 0
            for info in infos:
                if not re.fullmatch(r"Contents/section\d+\.xml", info.filename, re.I): continue
                root = safe_xml(zin.read(info.filename)); count = 0
                for n in root.iter():
                    if lname(n) == "t" and n.text and old in n.text:
                        c = n.text.count(old); n.text = n.text.replace(old, new); count += c; occurrences += c
                if count: patched[info.filename] = etree.tostring(root, xml_declaration=True, encoding="UTF-8", standalone=True)
            if not occurrences: raise ValueError(f"'{old}' 문자열을 찾지 못했습니다.")
            output.parent.mkdir(parents=True, exist_ok=True)
            with ZipFile(output, "w") as zout:
                for info in infos: zout.writestr(info, patched.get(info.filename, zin.read(info.filename)))
        after = self.parse(output)
        return {
            "occurrences": occurrences, "sourceUnchanged": sha256(source) == before_hash,
            "structurePreserved": len(before["paragraphs"]) == len(after["paragraphs"]) and len(before["tables"]) == len(after["tables"]),
            "output": str(output),
        }
