def documents_directory() -> Path:
    if sys.platform == "win32":
        try:
            from win32com.shell import shell, shellcon
            value = shell.SHGetFolderPath(0, shellcon.CSIDL_PERSONAL, 0, 0)
            if value:
                return Path(value)
        except Exception:
            pass
    return Path.home() / "Documents"


def next_hwpx_path() -> Path:
    folder = documents_directory() / "DocumentAssistant"
    folder.mkdir(parents=True, exist_ok=True)
    base = f"새문서_{datetime.now():%Y%m%d_%H%M%S}"
    path = folder / f"{base}.hwpx"
    n = 1
    while path.exists():
        path = folder / f"{base}_{n:02d}.hwpx"; n += 1
    return path


def validate_hwpx_file(path: Path) -> bool:
    try:
        with ZipFile(path) as zf:
            if "mimetype" not in zf.namelist(): return False
            mt = zf.read("mimetype").decode("utf-8", "ignore").lower()
            return "hwp" in mt and "zip" in mt and zf.testzip() is None
    except Exception:
        return False
